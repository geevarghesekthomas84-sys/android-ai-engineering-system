## Lab 1: Hardcoded Encryption Keys

**Learning Objective**: Demonstrate why hardcoded keys in source code provide no real security and how to use the Android Keystore for key management.

**OWASP Mobile**: M5 — Insufficient Cryptography

### Vulnerable Implementation

```kotlin
// SECURITY LAB: INTENTIONAL vulnerability — hardcoded encryption key
// VULNERABILITY: CWE-321 — Use of hard-coded cryptographic key

object InsecureCrypto {

    // VULNERABILITY: Key is embedded in the APK — extractable with jadx or strings
    private val ENCRYPTION_KEY = "MySuperSecretKey".toByteArray()  // 16 bytes for AES-128
    private val FIXED_IV = "1234567890abcdef".toByteArray()        // VULNERABILITY: static IV

    fun encrypt(plaintext: String): ByteArray {
        val cipher = Cipher.getInstance("AES/CBC/PKCS5Padding")
        val keySpec = SecretKeySpec(ENCRYPTION_KEY, "AES")
        val ivSpec = IvParameterSpec(FIXED_IV)  // VULNERABILITY: reused IV
        cipher.init(Cipher.ENCRYPT_MODE, keySpec, ivSpec)
        return cipher.doFinal(plaintext.toByteArray())
    }

    fun decrypt(ciphertext: ByteArray): String {
        val cipher = Cipher.getInstance("AES/CBC/PKCS5Padding")
        val keySpec = SecretKeySpec(ENCRYPTION_KEY, "AES")
        val ivSpec = IvParameterSpec(FIXED_IV)
        cipher.init(Cipher.DECRYPT_MODE, keySpec, ivSpec)
        return String(cipher.doFinal(ciphertext))
    }
}
```

### Exploitation Steps

```bash
# 1. Decompile the APK with jadx
jadx -d output/ app-vulnerable-debug.apk

# 2. Search for the hardcoded key
grep -r "MySuperSecretKey" output/
# Found in: output/sources/com/securitylab/crypto/InsecureCrypto.java

# 3. With the key and IV, decrypt any stored data:
python3 -c "
from Crypto.Cipher import AES
import base64
key = b'MySuperSecretKey'
iv  = b'1234567890abcdef'
ct  = base64.b64decode('<base64 ciphertext from the app>')
cipher = AES.new(key, AES.MODE_CBC, iv)
print(cipher.decrypt(ct))
"
```

### Remediation

```kotlin
// FIX: Use Android Keystore to generate and store keys in hardware
object SecureCrypto {

    private const val KEY_ALIAS = "lab_encryption_key"

    // FIX: Key is generated inside the Keystore and never leaves it
    private fun getOrCreateKey(): SecretKey {
        val keyStore = KeyStore.getInstance("AndroidKeyStore").apply { load(null) }
        keyStore.getEntry(KEY_ALIAS, null)?.let {
            return (it as KeyStore.SecretKeyEntry).secretKey
        }

        val keyGen = KeyGenerator.getInstance(KeyProperties.KEY_ALGORITHM_AES, "AndroidKeyStore")
        keyGen.init(
            KeyGenParameterSpec.Builder(KEY_ALIAS,
                KeyProperties.PURPOSE_ENCRYPT or KeyProperties.PURPOSE_DECRYPT)
                .setBlockModes(KeyProperties.BLOCK_MODE_GCM)
                .setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE)
                .setKeySize(256)
                .build()
        )
        return keyGen.generateKey()
    }

    // FIX: GCM mode with random IV — each encryption produces a unique ciphertext
    fun encrypt(plaintext: String): ByteArray {
        val cipher = Cipher.getInstance("AES/GCM/NoPadding")
        cipher.init(Cipher.ENCRYPT_MODE, getOrCreateKey())
        val iv = cipher.iv  // FIX: Random IV generated by the cipher
        val ciphertext = cipher.doFinal(plaintext.toByteArray())
        return iv + ciphertext  // Prepend IV for decryption
    }

    fun decrypt(data: ByteArray): String {
        val iv = data.copyOfRange(0, 12)  // GCM IV is 12 bytes
        val ciphertext = data.copyOfRange(12, data.size)
        val cipher = Cipher.getInstance("AES/GCM/NoPadding")
        cipher.init(Cipher.DECRYPT_MODE, getOrCreateKey(), GCMParameterSpec(128, iv))
        return String(cipher.doFinal(ciphertext))
    }
}
```

## Lab 2: Weak Algorithms (DES and MD5)

**Learning Objective**: Show why DES (56-bit key) and MD5 (for password hashing) are cryptographically broken.

### Vulnerable Implementation

```kotlin
// SECURITY LAB: INTENTIONAL vulnerability — weak cryptographic algorithms
// VULNERABILITY: CWE-327 — Use of a broken or risky cryptographic algorithm

object WeakCrypto {

    // VULNERABILITY: DES has a 56-bit key — brutable in hours on modern hardware
    fun encryptWithDES(plaintext: String, key: String): ByteArray {
        val cipher = Cipher.getInstance("DES/ECB/PKCS5Padding")  // VULNERABILITY: DES + ECB
        val keySpec = SecretKeySpec(key.toByteArray().copyOf(8), "DES")
        cipher.init(Cipher.ENCRYPT_MODE, keySpec)
        return cipher.doFinal(plaintext.toByteArray())
    }

    // VULNERABILITY: MD5 is collision-prone and far too fast for password hashing
    fun hashPassword(password: String): String {
        val digest = MessageDigest.getInstance("MD5")
        val hash = digest.digest(password.toByteArray())
        return hash.joinToString("") { "%02x".format(it) }
    }
}
```

### Exploitation

```bash
# MD5 hashes can be reversed via rainbow tables:
# hashPassword("password1") -> "7c6a180b36896a65a73e3a32"
# Look up on any MD5 reversal database (lab-local only) — instant result.

# DES brute-force: use a local tool to enumerate the 2^56 keyspace.
# Modern GPUs can test billions of DES keys per second.
```

### Remediation

```kotlin
// FIX: Use AES-256-GCM for encryption and Argon2/PBKDF2 for password hashing
object StrongCrypto {

    // FIX: AES-256-GCM via Android Keystore (see Lab 1 remediation)

    // FIX: PBKDF2 with high iteration count for password hashing
    fun hashPassword(password: String): String {
        val salt = ByteArray(32).also { SecureRandom().nextBytes(it) }
        val spec = PBEKeySpec(password.toCharArray(), salt, 120_000, 256)
        val factory = SecretKeyFactory.getInstance("PBKDF2WithHmacSHA256")
        val hash = factory.generateSecret(spec).encoded
        // Store salt and hash together
        return Base64.encodeToString(salt, Base64.NO_WRAP) + ":" +
               Base64.encodeToString(hash, Base64.NO_WRAP)
    }
}
```

## Lab 3: ECB Mode Encryption

**Learning Objective**: Visually demonstrate how ECB mode preserves plaintext patterns in ciphertext.

### Vulnerable Implementation

```kotlin
// SECURITY LAB: INTENTIONAL vulnerability — ECB mode
// VULNERABILITY: CWE-327 — ECB mode reveals patterns in plaintext

fun encryptECB(plaintext: ByteArray, key: SecretKey): ByteArray {
    // VULNERABILITY: ECB encrypts each block independently — identical plaintext
    // blocks produce identical ciphertext blocks, revealing data patterns
    val cipher = Cipher.getInstance("AES/ECB/PKCS5Padding")
    cipher.init(Cipher.ENCRYPT_MODE, key)
    return cipher.doFinal(plaintext)
}
```

### Exploitation Demonstration

```kotlin
// Encrypt a bitmap image using ECB mode:
// Identical pixel blocks produce identical cipher blocks.
// The output "encrypted" image still shows the silhouette of the original.
// This is the classic "ECB penguin" demonstration.

fun demonstrateECBWeakness(inputBitmap: ByteArray, key: SecretKey) {
    val ecbEncrypted = encryptECB(inputBitmap, key)
    // Save ecbEncrypted as a BMP — the image structure is visually apparent

    val cbcEncrypted = encryptCBC(inputBitmap, key)
    // Save cbcEncrypted as a BMP — appears as random noise (correct behavior)
}
```

### Remediation

```kotlin
// FIX: Use GCM mode (authenticated encryption) with a random IV
fun encryptGCM(plaintext: ByteArray, key: SecretKey): ByteArray {
    val cipher = Cipher.getInstance("AES/GCM/NoPadding")
    cipher.init(Cipher.ENCRYPT_MODE, key)
    val iv = cipher.iv  // FIX: Random 12-byte IV
    val ciphertext = cipher.doFinal(plaintext)
    return iv + ciphertext
}
```

## Lab 4: Predictable Initialization Vectors

**Learning Objective**: Show how a static or predictable IV combined with the same key allows ciphertext comparison attacks.

### Vulnerable Implementation

```kotlin
// SECURITY LAB: INTENTIONAL vulnerability — predictable IV
// VULNERABILITY: CWE-329 — Generation of predictable IV

fun encryptWithPredictableIV(plaintext: String, key: SecretKey): ByteArray {
    // VULNERABILITY: IV is derived from a counter — predictable and reused per value
    val counter = System.currentTimeMillis() / 1000  // Seconds granularity
    val iv = ByteArray(16)
    ByteBuffer.wrap(iv).putLong(counter)

    val cipher = Cipher.getInstance("AES/CBC/PKCS5Padding")
    cipher.init(Cipher.ENCRYPT_MODE, key, IvParameterSpec(iv))
    return iv + cipher.doFinal(plaintext.toByteArray())
}
```

### Remediation

```kotlin
// FIX: Always use a cryptographically random IV
fun encryptWithRandomIV(plaintext: String, key: SecretKey): ByteArray {
    val iv = ByteArray(12)
    SecureRandom().nextBytes(iv)  // FIX: Cryptographically random IV

    val cipher = Cipher.getInstance("AES/GCM/NoPadding")
    cipher.init(Cipher.ENCRYPT_MODE, key, GCMParameterSpec(128, iv))
    val ciphertext = cipher.doFinal(plaintext.toByteArray())
    return iv + ciphertext
}
```

## Lab 5: Insecure Random Number Generation

**Learning Objective**: Demonstrate how `java.util.Random` produces predictable sequences when used for security-sensitive values.

### Vulnerable Implementation

```kotlin
// SECURITY LAB: INTENTIONAL vulnerability — insecure PRNG
// VULNERABILITY: CWE-330 — Use of insufficiently random values

fun generateSessionToken(): String {
    // VULNERABILITY: java.util.Random is a linear congruential generator
    // Given a few outputs, the internal state can be recovered and future tokens predicted
    val random = Random(System.currentTimeMillis())  // VULNERABILITY: predictable seed
    val token = (1..32).map {
        "abcdefghijklmnopqrstuvwxyz0123456789"[random.nextInt(36)]
    }.joinToString("")
    return token
}

fun generateOTP(): String {
    // VULNERABILITY: Predictable 6-digit OTP
    val random = Random()
    return String.format("%06d", random.nextInt(1_000_000))
}
```

### Remediation

```kotlin
// FIX: Use SecureRandom for all security-sensitive random values
fun generateSessionToken(): String {
    val random = SecureRandom()  // FIX: Cryptographically secure PRNG
    val bytes = ByteArray(32)
    random.nextBytes(bytes)
    return Base64.encodeToString(bytes, Base64.URL_SAFE or Base64.NO_WRAP)
}

fun generateOTP(): String {
    val random = SecureRandom()  // FIX: Unpredictable OTP
    return String.format("%06d", random.nextInt(1_000_000))
}
```

## Lab 6: Custom Crypto Implementation

**Learning Objective**: Show why implementing your own cryptographic algorithm is almost always wrong, even when the idea seems clever.

### Vulnerable Implementation

```kotlin
// SECURITY LAB: INTENTIONAL vulnerability — homebrew encryption
// VULNERABILITY: CWE-327 — Custom crypto is trivially reversible

object HomebrewCrypto {

    // VULNERABILITY: Simple XOR "cipher" with a short repeating key
    fun encrypt(plaintext: String, key: String): ByteArray {
        return plaintext.toByteArray().mapIndexed { i, b ->
            (b.toInt() xor key[i % key.length].code).toByte()
        }.toByteArray()
    }

    // VULNERABILITY: ROT13-style substitution for "obfuscation"
    fun obfuscate(input: String): String {
        return input.map { c ->
            when (c) {
                in 'a'..'z' -> 'a' + (c - 'a' + 13) % 26
                in 'A'..'Z' -> 'A' + (c - 'A' + 13) % 26
                else -> c
            }
        }.joinToString("")
    }

    // VULNERABILITY: Base64 is encoding, not encryption
    fun "secure"Encode(data: String): String {
        return Base64.encodeToString(data.toByteArray(), Base64.DEFAULT)
    }
}
```

### Exploitation

```bash
# XOR with a short key is trivially broken with frequency analysis or known plaintext
# If we know one plaintext-ciphertext pair, we recover the key:
# key[i] = plaintext[i] XOR ciphertext[i]

# ROT13 is its own inverse:
echo "Cnffjbeq123" | tr 'A-Za-z' 'N-ZA-Mn-za-m'
# Output: Password123

# Base64 is decoded instantly:
echo "UGFzc3dvcmQxMjM=" | base64 -d
# Output: Password123
```

### Remediation

```kotlin
// FIX: Use standard, well-audited cryptographic libraries
// For encryption: AES-256-GCM via Android Keystore (see Lab 1)
// For password hashing: PBKDF2 or Argon2 (see Lab 2)
// For data integrity: HMAC-SHA256
// For key exchange: ECDH with Curve25519
// NEVER implement your own cryptographic algorithm
```
